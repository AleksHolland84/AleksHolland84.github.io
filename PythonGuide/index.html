import math

def pris_rosa(x):
    return math.floor(x / 75) * 50 + 200

def pris_frederik(x):
    return 1.5 * x

def billigst(a: list, b: list) -> list:
    name = a if a[1] < b[1] else b
    return name 


pris_mindst = None
for x in range(1,300):    
    name = billigst(
        ["Rosa", pris_rosa(x)],
        ["Frederik", pris_frederik(x)]
    )
    if not pris_mindst:
        pris_mindst = name[0]
        print(f'Billigst ved {name[0]}\'s julemarked, til {name[1]}kr., på {x} cm')
    elif name[0] != pris_mindst:
        pris_mindst = name[0]
        print(f'Billigst ved {name[0]}\'s julemarked, til {name[1]}kr., på {x} cm')

    result = (h*l*b)
    if result != 0:
        return (h*l*b < 225000)
